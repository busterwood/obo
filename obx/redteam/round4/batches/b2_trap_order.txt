    let xs = [1, 2]

=== b06_ifx_ternary_trap_order
// exit: 10
fn f(a: int, b: int) { println(a + b) }
fn main() {
    let xs = [1, 2]
    var z = 0
    let c = true
    f(if c { 10 / z } else { 0 }, xs[5])
}
